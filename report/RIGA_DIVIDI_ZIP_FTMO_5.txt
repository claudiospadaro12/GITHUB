& { $ErrorActionPreference='Stop'; $d=Join-Path ([Environment]::GetFolderPath('Desktop')) 'ABTG_RISULTATI'; $zs=Get-ChildItem -LiteralPath $d -Filter 'EXPORT_M1_FTMO_2*.zip' -File | Where-Object { $_.Name -notmatch 'PARTE' } | Sort-Object LastWriteTime -Descending | Select-Object -First 1; if(-not $zs){ throw ('Nessuno zip EXPORT_M1_FTMO_2*.zip in ' + $d) }; $np=5; Write-Host ('Zip di partenza: ' + $zs.FullName + ' (' + [math]::Round($zs.Length/1MB,1) + ' MB)') -ForegroundColor Cyan; $t=Join-Path $env:TEMP ('ftmo_split_' + (Get-Date).ToString('yyyyMMddHHmmss')); New-Item -ItemType Directory -Force -Path $t | Out-Null; Expand-Archive -LiteralPath $zs.FullName -DestinationPath (Join-Path $t 'src') -Force; $all=@(Get-ChildItem -LiteralPath (Join-Path $t 'src') -Recurse -File); $csv=@($all | Where-Object { $_.Extension -eq '.csv' } | Sort-Object Name); $oth=@($all | Where-Object { $_.Extension -ne '.csv' }); if($csv.Count -lt $np){ throw ('Trovati solo ' + $csv.Count + ' file CSV: non si divide in ' + $np) }; $tot=($csv | Measure-Object Length -Sum).Sum; foreach($k in 1..$np){ New-Item -ItemType Directory -Force -Path (Join-Path $t ('p' + $k)) | Out-Null }; $acc=0; $k=1; foreach($f in $csv){ if($k -lt $np -and $acc -ge ($tot*$k/$np)){ $k=$k+1 }; Copy-Item -LiteralPath $f.FullName -Destination (Join-Path $t ('p' + $k)); $acc=$acc+$f.Length }; foreach($f in $oth){ Copy-Item -LiteralPath $f.FullName -Destination (Join-Path $t 'p1') -Force }; $out=@(); foreach($k in 1..$np){ $zp=Join-Path $d ($zs.BaseName + '_PARTE' + $k + '.zip'); Remove-Item -LiteralPath $zp -Force -ErrorAction SilentlyContinue; Compress-Archive -Path (Join-Path (Join-Path $t ('p' + $k)) '*') -DestinationPath $zp -Force; $out += Get-Item -LiteralPath $zp }; Remove-Item -LiteralPath $t -Recurse -Force -ErrorAction SilentlyContinue; Write-Host '#########  ZIP DA ALLEGARE ALLA CHAT (5 PARTI)  #########' -ForegroundColor Black -BackgroundColor Yellow; foreach($o in $out){ $mb=[math]::Round($o.Length/1MB,1); $c=$(if($mb -gt 30){ 'Red' } else { 'Yellow' }); Write-Host ($o.FullName + '   ' + $mb + ' MB') -ForegroundColor Black -BackgroundColor $c; if($mb -gt 30){ Write-Host '   ATTENZIONE: sopra 30 MB, dimmelo e lo divido in piu parti.' -ForegroundColor Red } }; Start-Process explorer.exe -ArgumentList ('/select,"' + $out[0].FullName + '"') }
